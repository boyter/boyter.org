{{- /* Table render hook. Wraps every markdown table in a scroll container (so wide tables scroll instead of blowing out the page) and right aligns columns that hold only numbers, which is what most tables here are measurements. */ -}} {{- $numRE := `(?i)^\s*[+-]?[$€£]?\d[\d,_ ]*(\.\d+)?\s*(%|ms|µs|us|ns|s|m|h|kb|mb|gb|tb|b|x|k)?\s*$` -}} {{- $cols := 0 -}} {{- range .TBody -}} {{- if gt (len .) $cols }}{{ $cols = len . }}{{ end -}} {{- end -}} {{- /* A column is numeric when every non empty body cell in it looks like a number. */ -}} {{- $numeric := slice -}} {{- if gt $cols 0 -}} {{- range $i := seq 0 (sub $cols 1) -}} {{- $all := true -}} {{- $any := false -}} {{- range $.TBody -}} {{- if gt (len .) $i -}} {{- $text := trim (printf "%s" (index . $i).Text) " " -}} {{- if $text -}} {{- if findRE $numRE $text -}}{{ $any = true }}{{- else -}}{{ $all = false }}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $numeric = $numeric | append (and $all $any) -}} {{- end -}} {{- end -}} {{- /* Drop a header row that is entirely blank rather than render an empty band. */ -}} {{- $hasHead := false -}} {{- range .THead -}} {{- range . -}} {{- if trim (printf "%s" .Text) " " }}{{ $hasHead = true }}{{ end -}} {{- end -}} {{- end -}}
